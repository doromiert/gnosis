cmake_minimum_required(VERSION 3.21)
project(JupiterEngine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4 libadwaita-1 json-glib-1.0 epoxy webkitgtk-6.0)
set(CEF_ROOT "$ENV{CEF_ROOT}" CACHE PATH "CEF binary distribution")
if(CEF_WRAPPER_LIBRARY)
  add_library(libcef_dll_wrapper STATIC IMPORTED)
  set_target_properties(libcef_dll_wrapper PROPERTIES IMPORTED_LOCATION "${CEF_WRAPPER_LIBRARY}")
else()
  list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
  set(USE_SANDBOX OFF CACHE BOOL "Linux uses the Chromium sandbox")
  find_package(CEF REQUIRED)
  add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
endif()
set(JUPITER_BRIDGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/data/tab-bridge" CACHE PATH "Native extension tab bridge")
set(UBLOCK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../jupiter-prototype/.deps/uBlock0.chromium" CACHE PATH "Official full uBlock Origin")
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/resources.c"
  COMMAND "${GLIB_COMPILE_RESOURCES}" "${CMAKE_CURRENT_SOURCE_DIR}/data/engine.gresource.xml"
    "--sourcedir=${CMAKE_CURRENT_SOURCE_DIR}/data" --generate-source --c-name=jupiter_engine "--target=${CMAKE_CURRENT_BINARY_DIR}/resources.c"
  DEPENDS data/engine.gresource.xml data/clipboard.js data/extension-tabs.js)
add_library(jupiter-engine STATIC ephy-engine-view.cpp select/browser.cpp select/clipboard.cpp select/cursor.cpp
  select/policy.cpp select/extensions.cpp select/extension-host.cpp select/application.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/resources.c")
target_include_directories(jupiter-engine SYSTEM PUBLIC "${CEF_ROOT}")
target_include_directories(jupiter-engine PUBLIC . select)
target_link_libraries(jupiter-engine PUBLIC libcef_dll_wrapper "${CEF_ROOT}/Release/libcef.so" PkgConfig::GTK)
target_compile_definitions(jupiter-engine PUBLIC CEF_RESOURCE_DIR="${CEF_ROOT}/Resources"
  JUPITER_UBLOCK_DIR="${UBLOCK_DIR}" JUPITER_BRIDGE_DIR="${JUPITER_BRIDGE_DIR}")
if(BUILD_TESTING)
  add_executable(engine-view-test tests/engine-view-test.cpp)
  target_link_libraries(engine-view-test PRIVATE jupiter-engine)
  set_target_properties(engine-view-test PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}")
  add_executable(extension-probe tests/extension-probe.cpp)
  target_link_libraries(extension-probe PRIVATE jupiter-engine)
  set_target_properties(extension-probe PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}")
endif()
file(GLOB CEF_RUNTIME_FILES "${CEF_ROOT}/Release/*" "${CEF_ROOT}/Resources/*")
foreach(runtime_file IN LISTS CEF_RUNTIME_FILES)
  get_filename_component(runtime_name "${runtime_file}" NAME)
  file(CREATE_LINK "${runtime_file}" "${CMAKE_CURRENT_BINARY_DIR}/${runtime_name}" SYMBOLIC)
endforeach()
