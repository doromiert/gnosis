function(jupiter_wayland_protocol path output mode)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${output}"
    COMMAND "${WAYLAND_SCANNER}" "${mode}" "${WAYLAND_PROTOCOL_DIR}/${path}" "${CMAKE_CURRENT_BINARY_DIR}/${output}"
    DEPENDS "${WAYLAND_PROTOCOL_DIR}/${path}")
  set(COMPOSITOR_PROTOCOLS ${COMPOSITOR_PROTOCOLS} "${CMAKE_CURRENT_BINARY_DIR}/${output}" PARENT_SCOPE)
endfunction()
